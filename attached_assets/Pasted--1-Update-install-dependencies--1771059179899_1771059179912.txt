# 3. Custom Bash prompt
# -------------------------------
mv ~/.bashrc ~/.bashrc.bak.$(date +%s) 2>/dev/null
cat > ~/.bashrc << 'EOF'
# Prompt style Kali Linux
PS1='\[\e[1;32m\]┌──(\[\e[1;34m\]Termux㉿localhost\[\e[1;32m\])-[\[\e[1;37;1m\]\w\[\e[1;32m\]]\n\[\e[1;32m\]└─\[\e[1;34;1m\]$ \[\e[37;1m\]'

# Command not found handler
command_not_found_handle() {
    local PKG="$1"
    shift
    if pkg search "^${PKG}\$" 2>/dev/null | grep -q "^${PKG}/"; then
        printf "\033[1;32m[+] Installing %s...\033[0m\n" "$PKG"
        if pkg install -y "$PKG"; then
            if command -v "$PKG" >/dev/null 2>&1; then
                printf "\033[1;32m[✓] Installed successfully.\033[0m\n"
                "$PKG" "$@"
            else
                printf "\033[1;33m[!] Installed but no executable found.\033[0m\n"
            fi
        else
            printf "\033[1;31m[✗] Failed to install %s.\033[0m\n" "$PKG"
        fi
    else
        printf "\033[1;31m[!] Not found in repo: %s\033[0m\n" "$PKG"
    fi
}
EOF

# -------------------------------
# 4. Reset Semua ke default
# -------------------------------
reset_termux() {
    echo "⚠️  Reset Termux ke default..."
    # Restore bashrc
    if [ -f ~/.bashrc.bak.* ]; then
        cp $(ls -t ~/.bashrc.bak.* | head -n1) ~/.bashrc
        echo "✅ Bash prompt dikembalikan"
    fi
    # Hapus konfigurasi neofetch
    rm -rf ~/.config/neofetch
    echo "✅ Neofetch dihapus"
    # Hapus instalasi tambahan (opsional, bisa dicustom)
    # pkg uninstall -y lolcat neofetch ruby curl git make
    echo "✅ Paket tambahan bisa dihapus manual jika diinginkan"
    echo "✅ Selesai! Restart Termux"
}

echo "✅ Termux prompt + Neofetch setup complete — Restart Termux"
echo "ℹ️  Jika ingin reset semua ke default, jalankan: reset_termux"
